        * .addField(key:string, id:string, name:string, type:string, alt:string, changes:int, value:string) - Add a field to form
        * .setObfuscatedFields(obfuscatedFields:array) - Set Obfuscated Field Keys
        * .setPostprocess(postprocess:JSON) - set Post Process Config
        * .scan(tags:string...) - scan HTML Elements to generate fields by tags
        * .print() - print out CI360 custom event by console.log()
        * .send() - send event to CI360
        */
        var CI360FormSubmit = function(root, formID, formName){
            if(!root || !formID || !formName) return;
            this.count = 0;
            this.obfuscated = [];
            this.root = root;
            this.formName = formName;
            this.event = {
                'eventName': 'submit',
                'location': root,
                'form_o_id': formID,
                'form_h_id': formID,
                'form_h_name': formName,
                'id' : formID,
                'name' : formName
            };
            this.positive = false;
        };
        CI360FormSubmit.prototype.addField = function(key, id, name, type, alt, changes, value){
            if(!key || (!id && !name)) return;
            this.event['form.f.' + key + '._h.id'] = id;
            this.event['form.f.' + key + '._h.name'] = name;
            this.event['form.f.' + key + '._h.type'] = type;
            this.event['form.f.' + key + '._h.alt'] = alt;
            this.event['form.f.' + key + '._o.changes'] = changes;
            if(this.obfuscated.indexOf(key)>-1 || type==='password')
                this.event['form.f.' + key] = {'value':value, 'obscured':true};
            else
                this.event['form.f.' + key] = value;
        };
        CI360FormSubmit.prototype.setObfuscatedFields = function(obfuscatedFields){
            if(obfuscatedFields) {
                this.obfuscated = obfuscatedFields;
            }
        };
        CI360FormSubmit.prototype.setPostprocess = function(postprocess){
            if(postprocess) this.postprocess = postprocess;
        };
        CI360FormSubmit.prototype.scan = function(tags){
            this.event['form._o.attempts'] = ++this.count;
            //var elements = jQuery(this.root).find(tags);
            var elements = jQuery("#"+this.event['form_o_id']).find(tags);
            for(var j=0;j<elements.length;j++){
                var element = jQuery(elements[j]);
                var type = element[0].type || 'text';
                if(type==='radio' && !element[0].checked) continue; //No Select of RadioButton
                var value = '';
                switch(type){
                    case 'checkbox': value = element[0].checked; break;
                    case 'select-one': value = element.val(); break;
                    default: value = element.val() || element.text();
                }
                if(!value && type!=='checkbox') continue; //No Value;
                var fieldKey = element[0].id || element[0].name || element[0].classList[0];
                if(!fieldKey){ continue; /*No ID and Name and Class*/ }else{ fieldKey = fieldKey.split('.').join('_') + ((type==='checkbox')?'_'+element.val():''); }
                var id = element[0].id || element[0].name || element[0].classList[0];
                var name = element[0].name || element[0].id || element[0].classList[0];
                var alt = element[0].alt || '';
                this.addField(fieldKey, id, name, type, alt, this.count, value);
            }
        };

            console.log(this.event);
        };

            this.positive = positive;
        };

            CI360GenericEvents.add({event:this.event, positive:this.positive, postprocess:this.postprocess}, this.postprocess);

        /**
        * CI360 Custom Event Binding
        * customClick(clickName:string, clickSelector:string, config:JSON, pathNameRegex:string(optional)) - Bind Click Event
                -> config:JSON { AttributeMappings:JSON(optional), ReplaceName:boolean }
        */
        var customClick = function(clickName, clickSelector, config, pathNameRegex){
            try{
                pathNameRegex = UAT_DOMAIN_CONVERT(pathNameRegex);
                if(pathNameRegex && !eval('/^' + pathNameRegex.split('/').join('\\/') + '$/').test(location.pathname)) return;
                var root = jQuery(clickSelector); if(root.length===0) return;
                var isName = "" == clickName?true:false;
                for(var i=0;i<root.length;i++){
                    try{
                        debug('===============', 'Click Object "' + clickName + '" is Detected.', root[i]);
                        if(config.ReplaceName) {
                            debug('Click Name Replaced: ' + jQuery(root[i]).attr('name') + ' -> ' + clickName);
                            jQuery(root[i]).attr('name',clickName);
                            if(jQuery(root[i]).attr('id')==null){
                                jQuery(root[i]).attr('id',clickName);
                            }
                        }
                        if (isName) {
                            if (jQuery(root[i]).attr('title')!=null) {
                                jQuery(root[i]).attr('name',jQuery(root[i]).attr('title'));
                                jQuery(root[i]).attr('id',jQuery(root[i]).attr('title'));
                                clickName = jQuery(root[i]).attr('title');
                            } else {
                                clickName = jQuery(root[i]).attr('id');
                                jQuery(root[i]).attr('name',jQuery(root[i]).attr('id'));
                            }
                        }
                        debug('Button Click Event Binding.');
                        jQuery(root[i]).on('click',function(e){
                            var isDisabled = $(e.currentTarget).hasClass('disable');
                            if(isDisabled) return;
                            debug('Click Event Triggered.');
                            jsPerformance.start('ClickEventSent');
                            if (isName) clickName = $(this).attr('name');
                            var clickObj = new CI360Click(clickName, clickName, e.target);
                            clickObj.scan(config.AttributeMappings);
                            clickObj.event.target_selector_path = $(this).getPath();
                            // by 「樂e購」：會員、非會員及企業，click event
                            if ("CurrencyBtn" == clickName) { // by 官網：旅客資訊是否有輸入會員號碼，click event
                                clickObj.event.target_inner_text = "N";
                                jQuery("input[name='FREQ_Number_0']").each(function(){
                                    if ($(this).val().length > 0) {
                                        clickObj.event.target_inner_text = "Y";
                                    }
                                });
                            } else if ("mEvaair" == clickName) { //手機版訂票網，進入step2無page load，強制送出
                                clickObj.event.target_inner_text = "ALPI";
                            } else if (config.Country && location.pathname.indexOf('booking-multi.aspx')==-1 ) { // 判斷是否為美加
                                var depCountry = "";
                                var arrCountry = "";
                                var depFlight = "";
                                var arrFlight = "";
                                var ckin_cabin = "";
                                if (jQuery('select#content_ddl_cabinclass').length > 0){
                                    ckin_cabin = jQuery('select#content_ddl_cabinclass').val();
                                } else {
                                    ckin_cabin = jQuery('select#booking_online_ddl_cabinclass').val();
                                }
                                if (jQuery("#txt_From").length > 0) {
                                    depCountry = getCountryCode(jQuery("#txt_From").val().split(" - ")[1]);
                                    arrCountry = getCountryCode(jQuery("#txt_To").val().split(" - ")[1]);
                                    depFlight = jQuery("#txt_From").val().split(" - ")[1] ;
                                    arrFlight = jQuery("#txt_To").val().split(" - ")[1] ;
                                } else if (jQuery("#content_txt_From").length > 0) {
                                    depCountry = getCountryCode(jQuery("#content_txt_From").val().split(" - ")[1]);
                                    arrCountry = getCountryCode(jQuery("#content_txt_To").val().split(" - ")[1]);
                                    depFlight = jQuery("#content_txt_From").val().split(" - ")[1] ;
                                    arrFlight = jQuery("#content_txt_To").val().split(" - ")[1] ;
                                } else if (jQuery("#booking_online_txt_From").length > 0) {
                                    depCountry = getCountryCode(jQuery("#booking_online_txt_From").val().split(" - ")[1]);
                                    arrCountry = getCountryCode(jQuery("#booking_online_txt_To").val().split(" - ")[1]);
                                    depFlight = jQuery("#booking_online_txt_From").val().split(" - ")[1] ;
                                    arrFlight = jQuery("#booking_online_txt_To").val().split(" - ")[1] ;
                                }
    
                                if ('US/CA/SV/PA'.indexOf(depCountry) != -1 || 'US/CA/SV/PA'.indexOf(arrCountry) != -1) {
                                    $.cookie("flightSearchCountry", "USA", {
                                        expires: 1,
                                        path: '/',
                                        domain: '.evaair.com'
                                    });
                                } else {
                                    $.cookie("flightSearchCountry", "NonUSA", {
                                        expires: 1,
                                        path: '/',
                                        domain: '.evaair.com'
                                    });
                                }
                                $.cookie("depFlight", depFlight, {
                                        expires: 1,
                                        path: '/',
                                        domain: '.evaair.com'
                                    });
                                $.cookie("arrFlight", arrFlight, {
                                        expires: 1,
                                        path: '/',
                                        domain: '.evaair.com'
                                    });
                                $.cookie("cabin", ckin_cabin, {
                                        expires: 1,
                                        path: '/',
                                        domain: '.evaair.com'
                                    });
                            } else if (config.Country && location.pathname.indexOf('booking-multi.aspx')!=-1 ) { // 多航班判斷是否為美加
                                var Country_m = [];
                                var depFlight_m = [] ;
                                var arrFlight_m = [] ;
                                if (jQuery("#content_txt_From1").length > 0) {
                                    for (var i=0 ; i<6 ;i++ ) {
                                        depFlight_m[i]=jQuery("#content_txt_From"+String(i+1)).val();
                                        arrFlight_m[i]=jQuery("#content_txt_To"+String(i+1)).val();
                                        Country_m[i]=getCountryCode(depFlight_m[i].split(" - ")[1]);
                                        Country_m[i+6]=getCountryCode(arrFlight_m[i].split(" - ")[1]);
                                    }
                                }
                                Country_m = Country_m.filter(function(e){return e});
    
                                for (var j=0 ; j<Country_m.length ;j++)
                                {
                                    if ('US/CA/SV/PA'.indexOf(Country_m[j]) != -1)
                                    {
                                        $.cookie("flightSearchCountry", "USA", {
                                        expires: 1,
                                        path: '/',
                                        domain: '.evaair.com'
                                        });
                                        break ;
                                    }  else {
                                        $.cookie("flightSearchCountry", "NonUSA", {
                                        expires: 1,
                                        path: '/',
                                        domain: '.evaair.com'
                                        });
                                    }
                                }
                            }
                            else if (config.ReplaceInnerText) {
                                clickObj.event.target_inner_text = clickName;
                            }
                            if(config.Positive) clickObj.setPositive(config.Positive);
                            if(debug_mode) clickObj.print();
                            clickObj.send();
                            jsPerformance.stop('ClickEventSent');
                        });
                    }catch(e){
                        console.error('customClick[' + i + '](' + clickName + ',' + JSON.stringify(config) + ',' + pathNameRegex + ') got error: ' + e);
                    }
                }
            }catch(e){
                console.error('customClick(' + clickName + ',' + clickSelector + ',' + JSON.stringify(config) + ',' + pathNameRegex + ') got error: ' + e);
            }
        };
